import numpy as np
import matplotlib.pyplot as plt

# Domain
x = np.linspace(0, 1, 2000)

# Example 1: Compact candidate family (smooth functions)
def f_family_smooth(n, x):
    return np.sin(np.pi * x) / (1 + n)  # smooth, shrinking amplitude

# Example 2: Non-compact family (high oscillations)
def f_family_oscillatory(n, x):
    return np.sin(n * np.pi * x)  # oscillates faster with n

# L2 norm helper
def L2_norm(f, g, x):
    return np.sqrt(np.trapz((f - g)**2, x))

# Translation invariance test
def translation_norms(family, max_n=5, hs=np.linspace(0,0.1,20)):
    norms = []
    for h in hs:
        vals = []
        for n in range(1, max_n+1):
            f = family(n, x)
            f_shift = family(n, (x+h) % 1)  # periodic shift
            vals.append(L2_norm(f, f_shift, x))
        norms.append(max(vals))  # sup over family subset
    return hs, norms

# Compute for both families
hs, norms_smooth = translation_norms(f_family_smooth)
_, norms_osc = translation_norms(f_family_oscillatory)

# Plot results
plt.figure(figsize=(8,5))
plt.plot(hs, norms_smooth, marker='o', label="Smooth shrinking family")
plt.plot(hs, norms_osc, marker='s', label="Oscillatory family")
plt.title("Frechet Kolmogorov Criterion Translation Equicontinuity")
plt.xlabel("Shift h")
plt.ylabel(r"$\sup_n || f(x+h)-f(x) ||_{L^2}$")
plt.legend()
plt.grid(True)
plt.show()
